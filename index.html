async function loadDetails(item){ const overviewEl = document.getElementById("modalOverview"); const box = document.getElementById("episodeControls"); box.innerHTML = ""; try { // ================= MOVIE ================= if(item.type === "movie"){ box.style.display = "none"; const res = await fetch( `https://api.themoviedb.org/3/movie/${item.id}?api_key=${TMDB_API_KEY}` ); const data = await res.json(); overviewEl.textContent = data.overview || "No description available."; return; } // ================= ANIME ================= if(item.type === "anime"){ box.style.display = "block"; const res = await fetch("https://graphql.anilist.co", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ query: ` query ($id: Int) { Media(id: $id, type: ANIME) { description(asHtml: false) } } `, variables: { id: item.id } }) }); const json = await res.json(); const media = json?.data?.Media; const clean = media?.description ? media.description.replace(//gi, "\n") : "No description available."; overviewEl.textContent = clean; renderAnimeControls(item.id); return; } // ================= TV ================= box.style.display = "block"; const res = await fetch( `https://api.themoviedb.org/3/tv/${item.id}?api_key=${TMDB_API_KEY}` ); const data = await res.json(); overviewEl.textContent = data.overview || "No description available."; renderTVControls(item.id); } catch (e){ console.error(e); overviewEl.textContent = "No description available."; box.style.display = "none"; } }
